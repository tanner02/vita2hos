cmake_minimum_required(VERSION 3.20)
project(vita2hos_shader_integration LANGUAGES C CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(fmt REQUIRED)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/external/Vita3K/external/glslang" glslang)
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/external/Vita3K/external/SPIRV-Cross" spirv-cross)
add_library(util INTERFACE)
target_include_directories(util INTERFACE stubs "${REPO}/include/vita3k"
    "${REPO}/external/Vita3K/vita3k/util/include" "${REPO}/external/vita-headers/include")
target_link_libraries(util INTERFACE fmt::fmt)
add_library(features INTERFACE)
target_include_directories(features INTERFACE "${REPO}/external/Vita3K/vita3k/features/include")
add_subdirectory("${REPO}/external/Vita3K/vita3k/gxm" gxm)
add_subdirectory("${REPO}/external/Vita3K/vita3k/shader" shader)
include("${REPO}/cmake/Vita3KUniformShuffle.cmake")
vita2hos_fix_uniform_shuffle("${REPO}")
add_executable(shader_integration main.cpp
    "${REPO}/source/vita3k_shader_recompiler_iface_c.cpp" "${REPO}/source/gxm_uniform_layout.c")
target_include_directories(shader_integration PRIVATE "${REPO}/include"
    "${REPO}/external/Vita3K/external/glslang")
target_link_libraries(shader_integration PRIVATE shader spirv-cross-glsl)
enable_testing()
add_test(NAME real_gxp_layouts COMMAND shader_integration
    "${REPO}/external/Vita3K/tools/native-tool/src/shaders")
